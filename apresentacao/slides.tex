\begin{frame}{Ideia 2: o desempate lexicográfico e o seu ganho exato}
  \begin{block}{Regra}
    Ordene pelos votos dos juízes; entre opiniões com o mesmo número de votos, ordene pelo sinal barato.
  \end{block}
  \vspace{0.2cm}
  \begin{exampleblock}{Decomposição (elementar) do AUROC}
    $$\AUC(\text{juízes} \lex \text{sinal}) \;=\; \AUC(\text{juízes}) \;+\; \tau\,\Big(\alpha - \tfrac{1}{2}\Big)$$
    \small $\tau$: fração de pares empatados (vem só dos votos). \quad
    $\alpha$: AUROC do sinal barato \textbf{dentro dos empates}.
  \end{exampleblock}
  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\centering Comitê de 12 juízes}\centering
        $0{,}9260 \to \mathbf{0{,}9300}$\\ \small acima do comitê em 20 de 20 partições (stacking: 1 de 20)
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\centering O que importa é $\alpha$}\centering\small
        A AUROC global do sinal barato não entra na fórmula.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{O que não funcionou e os limites}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Resultados negativos (reportados)}\small
        \begin{itemize}\setlength{\itemsep}{3pt}
          \item 5 features estruturais (homologia, Hodge\ldots): nada acrescentam ao comitê.
          \item 6 tentativas de melhorar $\alpha$: todas pioram.
          \item Com 12 juízes, a fila de revisão não melhora de forma mensurável.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Limites}\small
        \begin{itemize}\setlength{\itemsep}{3pt}
          \item Rótulos do PublicHearingBR usados como dados.
          \item Ganho sobre os 12 juízes é pequeno (+0,004), mas consistente.
          \item Onde o comitê erra, o sinal barato erra junto (0,47).
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
